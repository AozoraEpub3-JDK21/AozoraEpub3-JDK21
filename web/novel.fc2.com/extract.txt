## 【2026-08-11 の構造変更】
##   作品タイトルの見出しが .sh_heading_main_a → <h2 class="default_page_title"> に変更され、
##   「SERIES/TITLE : タイトルがありません」で変換不能になっていた。
##   それ以外のセレクタ (.username / .novel_comment / .novel_img / .noveldescription /
##   .novel_first / .novel_maincontent 系) は 2026-08-11 時点の HTML でも健在。
##   旧セレクタはキャッシュ済み HTML のために候補として残してある。
##
## 【変換モデル】
##   目次 (mode=tc) の「最初から読む」(.novel_first a) を起点に、PAGE_URL で
##   pg=1..PAGE_NUM (.noveldescription のページ数、例: 441) を順に取得する。
##   各ページの中見出しは .novel_subtitle (無いページでは出力されない)

## 一覧ページの抽出設定
## タイトル
## 新: <h2 class="default_page_title"> / 旧: .sh_heading_main_a (キャッシュ用に残置)
TITLE	.default_page_title:0,.sh_heading_main_a:0
## 著者
AUTHOR	.username a:0,.username:0
## 説明
DESCRIPTION	.novel_comment:0
## 表紙画像
COVER_IMG	.novel_img img:0

## 更新日時 比較文字列 (hrefに1:1で対応するタグ 内部のinnerHTML文字列で比較)
UPDATE	.noveldescription table td p:0

################
## 各話へのリンク
##HREF	a	^\/novel\.php\?mode=rd\&

################
## ページ番号 数字のみ利用 (前優先 ^[^\d]*(\d+) )
PAGE_NUM	.noveldescription table td p:2	^[^\d]*(\d+).*$	$1
## ページ番号方式のベースURL ページ番号をURLの後ろにタブで結合した文字列の置換パターン
PAGE_URL	.novel_first a:0	^(.*pg=)\d+\t(\d+)$	$1$2
